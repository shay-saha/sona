set(SONA_NEMOTRON_ROOT "${CMAKE_SOURCE_DIR}/research/cache/nemotron-build"
    CACHE PATH "Optional NeMo-Speech.cpp build")
set(SONA_SHERPA_ROOT "${CMAKE_SOURCE_DIR}/research/cache/conversation/sherpa-onnx-v1.13.8-linux-x64-shared-no-tts-lib"
    CACHE PATH "Optional sherpa-onnx 1.13.8 CPU runtime")
option(SONA_ENABLE_DIARIZATION "Enable the optional native conversation worker" ON)
if(SONA_ENABLE_DIARIZATION)
    unset(NEMOTRON_INCLUDE_DIR CACHE)
    unset(NEMOTRON_LIBRARY CACHE)
    find_path(NEMOTRON_INCLUDE_DIR nemo_speech/diar.h
        HINTS "${SONA_NEMOTRON_ROOT}/../nemotron-runtime/include" NO_DEFAULT_PATH)
    find_library(NEMOTRON_LIBRARY nemo_speech_asr_c
        HINTS "${SONA_NEMOTRON_ROOT}/bin" NO_DEFAULT_PATH)
    if(NEMOTRON_INCLUDE_DIR AND NEMOTRON_LIBRARY)
        target_include_directories(sona_stream SYSTEM PRIVATE ${NEMOTRON_INCLUDE_DIR})
        target_link_libraries(sona_stream PRIVATE ${NEMOTRON_LIBRARY})
        target_compile_definitions(sona_stream PRIVATE SONA_NEMOTRON)
        if(EXISTS "${SONA_NEMOTRON_ROOT}/CMakeCache.txt")
            file(STRINGS "${SONA_NEMOTRON_ROOT}/CMakeCache.txt" NEMOTRON_CUDA REGEX "^GGML_CUDA:BOOL=ON$")
            if(NEMOTRON_CUDA)
                target_compile_definitions(sona_stream PRIVATE SONA_NEMOTRON_CUDA)
            endif()
        endif()
    endif()
    find_path(SHERPA_INCLUDE_DIR sherpa-onnx/c-api/c-api.h
        HINTS "${SONA_SHERPA_ROOT}/include" "${SONA_SHERPA_ROOT}/../include" NO_DEFAULT_PATH)
    find_library(SHERPA_LIBRARY sherpa-onnx-c-api HINTS "${SONA_SHERPA_ROOT}/lib" NO_DEFAULT_PATH)
    if(SHERPA_INCLUDE_DIR AND SHERPA_LIBRARY)
        target_include_directories(sona_diarization_backend SYSTEM PRIVATE ${SHERPA_INCLUDE_DIR})
        target_link_libraries(sona_diarization_backend PRIVATE ${SHERPA_LIBRARY})
        target_compile_definitions(sona_diarization_backend PUBLIC SONA_SHERPA)
        if(BUILD_TESTING AND NEMOTRON_LIBRARY AND
           EXISTS "${CMAKE_SOURCE_DIR}/research/cache/conversation/Nemotron-3-Diarization.q8_0.gguf")
            find_package(Python3 COMPONENTS Interpreter)
            if(Python3_Interpreter_FOUND)
                add_test(NAME conversation_neural_protocol COMMAND ${Python3_EXECUTABLE}
                    ${CMAKE_SOURCE_DIR}/tests/stream_protocol.py $<TARGET_FILE:sona_stream>
                    ${CMAKE_SOURCE_DIR}/research/cache/conversation)
                set_tests_properties(conversation_neural_protocol PROPERTIES TIMEOUT 150)
            endif()
        endif()
        if(BUILD_TESTING AND EXISTS "${CMAKE_SOURCE_DIR}/research/cache/conversation/nemo_en_titanet_small.onnx")
            find_package(Python3 COMPONENTS Interpreter)
            if(Python3_Interpreter_FOUND)
                add_test(NAME conversation_native_worker COMMAND ${Python3_EXECUTABLE}
                    ${CMAKE_SOURCE_DIR}/tests/diarize_protocol.py $<TARGET_FILE:sona_diarize>
                    ${CMAKE_SOURCE_DIR}/research/cache/conversation)
                set_tests_properties(conversation_native_worker PROPERTIES TIMEOUT 90)
            endif()
        endif()
    endif()
endif()
if(BUILD_TESTING)
    add_executable(conversation_tests tests/conversation_tests.cpp)
    target_link_libraries(conversation_tests PRIVATE sona_conversation)
    add_test(NAME conversation_accounting COMMAND conversation_tests)
    if(UNIX)
