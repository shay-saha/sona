cmake_minimum_required(VERSION 3.21)
project(SONA VERSION 0.3.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_compile_options("$<$<CXX_COMPILER_ID:GNU,Clang>:-Wall;-Wextra;-Wpedantic>")
include(CTest)

add_library(sona_dsp src/analysis.cpp src/speaker_features.cpp)
target_include_directories(sona_dsp PUBLIC src)
if(BUILD_TESTING)
    add_executable(analysis_tests tests/analysis_tests.cpp)
    target_link_libraries(analysis_tests PRIVATE sona_dsp)
    add_test(NAME signal_analysis COMMAND analysis_tests)
endif()

add_library(sona_simulation src/simulation.cpp)
target_include_directories(sona_simulation PUBLIC src)
if(BUILD_TESTING)
    add_executable(simulation_tests tests/simulation_tests.cpp)
    target_link_libraries(simulation_tests PRIVATE sona_simulation)
    add_test(NAME surface_simulation COMMAND simulation_tests)
endif()

add_executable(sona_features src/features_main.cpp)
target_link_libraries(sona_features PRIVATE sona_dsp)
if(BUILD_TESTING)
    add_executable(speaker_features_tests tests/speaker_features_tests.cpp)
    target_link_libraries(speaker_features_tests PRIVATE sona_dsp)
    add_test(NAME speaker_descriptors COMMAND speaker_features_tests)
endif()
